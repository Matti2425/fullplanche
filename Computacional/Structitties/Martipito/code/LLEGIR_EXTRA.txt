function plotModes(plotName, Phi, freq, imodes)
    % Phi    : [Ndof x Nmodes]
    % freq   : [Nmodes x 1]
    % imodes : Vector of indices to plot
    
    % Load mesh data locally or pass it in. 
    % Since functions define scope, we assume xn, Tn are available via evalin 
    % or we need to pass them. For this template, I will assume variables exist 
    % in workspace (MATLAB script behavior) or use evalin. 
    % Better practice: Pass xn/Tn as arguments. I will fetch from base workspace.
    
    xn = evalin('base', 'xn');
    Tn = evalin('base', 'Tn');
    
    figure('Name', plotName, 'NumberTitle', 'off', 'Color', 'w');
    
    num_plots = length(imodes);
    rows = ceil(sqrt(num_plots));
    cols = ceil(num_plots/rows);
    
    for k = 1:num_plots
        mode_idx = imodes(k);
        subplot(rows, cols, k);
        
        mode_shape = Phi(:, mode_idx);
        
        % Normalize mode for plotting
        max_disp = max(abs(mode_shape(1:6:end).^2 + mode_shape(2:6:end).^2 + mode_shape(3:6:end).^2));
        scale = 0.1 * max(max(xn)) / sqrt(max_disp);
        
        % Extract deformation
        dx = mode_shape(1:6:end);
        dy = mode_shape(2:6:end);
        dz = mode_shape(3:6:end);
        
        xn_def = xn + scale * [dx, dy, dz];
        
        % Plot
        patch('Vertices', xn_def, 'Faces', Tn, ...
              'FaceVertexCData', dz, ... % Color by Z
              'FaceColor', 'interp', 'EdgeColor', 'k', 'FaceAlpha', 0.8);
          
        title(sprintf('Mode %d: %.2f Hz', mode_idx, freq(mode_idx)));
        axis equal; axis vis3d; axis off;
        view(3);
    end
    sgtitle(plotName);
    colormap jet;
end

function plotDeformed(plotName, xn, Tn, u, scale)
    % xn : [Nnodes x 3]
    % Tn : [Nelem x 4]
    % u  : [Ndof x 1]
    
    figure('Name', plotName, 'NumberTitle', 'off', 'Color', 'w');
    
    % Extract displacements
    ux = u(1:6:end);
    uy = u(2:6:end);
    uz = u(3:6:end);
    
    % Deformed coordinates
    xn_def = xn + scale * [ux, uy, uz];
    
    % Plot undeformed (wireframe, transparent)
    patch('Vertices', xn, 'Faces', Tn, ...
          'FaceColor', 'none', 'EdgeColor', [0.7 0.7 0.7], 'LineStyle', ':');
    hold on;
    
    % Plot deformed (colored by Z displacement)
    % CData is used for color mapping. We map Z-displacement to color.
    % We need one color value per vertex or per face.
    cdata = uz; 
    
    p = patch('Vertices', xn_def, 'Faces', Tn, ...
          'FaceVertexCData', cdata, ...
          'FaceColor', 'interp', 'EdgeColor', 'k', 'FaceAlpha', 0.9);
      
    title(sprintf('%s (Scale: %.1f)', plotName, scale));
    axis equal; axis vis3d;
    xlabel('X'); ylabel('Y'); zlabel('Z');
    colorbar; colormap jet;
    view(3); grid on;
end

%% --- Modal postprocess (Part II requirement): ubar_y, ubar_z, thetabar_x vs span ---

n_pts = length(indSpar1);
x_span = xn(indSpar1,1);
dist_y = (y2 - y1);

for k = 1:Nm   % or for k = 1:n_modes_calc if you prefer
    % Modal vector (full size Ndof)
    phi = Phi(:,k);

    ubar_y = zeros(n_pts,1);
    ubar_z = zeros(n_pts,1);
    thetabar_x = zeros(n_pts,1);

    for i = 1:n_pts
        n1 = indSpar1(i);   % front spar node index at station i
        n2 = indSpar2(i);   % rear spar node index at station i

        uy1 = phi(6*(n1-1) + 2);
        uy2 = phi(6*(n2-1) + 2);

        uz1 = phi(6*(n1-1) + 3);
        uz2 = phi(6*(n2-1) + 3);

        % Averages / twist (statement formulas)
        ubar_y(i) = 0.5*(uy1 + uy2);
        thetabar_x(i) = (uz2 - uz1) / dist_y;
        ubar_z(i) = uz1 + thetabar_x(i)*(yc - y1);
    end

    % Normalize for visualization (optional, same reason as Part I)
    scale = max(abs([ubar_y; ubar_z; thetabar_x])) + eps;
    ubar_y = ubar_y/scale;
    ubar_z = ubar_z/scale;
    thetabar_x = thetabar_x/scale;

    figure;
    subplot(3,1,1); plot(x_span, ubar_y, 'LineWidth', 1.5); grid on;
    ylabel('$\bar{u}_y$', 'Interpreter','latex');

    subplot(3,1,2); plot(x_span, ubar_z, 'LineWidth', 1.5); grid on;
    ylabel('$\bar{u}_z$', 'Interpreter','latex');

    subplot(3,1,3); plot(x_span, thetabar_x, 'LineWidth', 1.5); grid on;
    ylabel('$\bar{\theta}_x$', 'Interpreter','latex');
    xlabel('Span $x$ [m]', 'Interpreter','latex');

    sgtitle(sprintf('Mode %d (f = %.2f Hz)', k, freq(k)));
end

function plotModes_beam(plotName, xn, Tn, Phi, freq, imodes)
% Beam mode shape plot (shell-like style):
% - undeformed discretised line (grey dashed)
% - deformed line, coloured by u_z (jet)
%
% xn : [Nnodes x 3]
% Tn : [Nelem x 2] connectivity
% Phi: [Ndof x Nmodes] full eigenvectors (including fixed DOFs if you rebuilt them)
% freq: [Nmodes x 1]
% imodes: indices to plot

    figure('Name', plotName, 'NumberTitle','off', 'Color','w');

    num_plots = length(imodes);
    rows = ceil(sqrt(num_plots));
    cols = ceil(num_plots/rows);

    x0 = xn(:,1); y0 = xn(:,2); z0 = xn(:,3);

    for kk = 1:num_plots
        I = imodes(kk);
        subplot(rows, cols, kk); hold on;

        mode = Phi(:,I);

        % Extract translational components
        dx = mode(1:6:end);
        dy = mode(2:6:end);
        dz = mode(3:6:end);

        % Scale so it looks nice (eigenvectors arbitrary amplitude)
        max_disp = max(sqrt(dx.^2 + dy.^2 + dz.^2)) + eps;
        scale = 0.15 * (max(x0) - min(x0)) / max_disp;

        x = x0 + scale*dx;
        y = y0 + scale*dy;
        z = z0 + scale*dz;

        % --- Undeformed discretised beam (grey dashed) ---
        for e = 1:size(Tn,1)
            n1 = Tn(e,1); n2 = Tn(e,2);
            plot3([x0(n1) x0(n2)], [y0(n1) y0(n2)], [z0(n1) z0(n2)], ...
                  ':', 'Color', 0.7*[1 1 1], 'LineWidth', 1.0);
        end

        % --- Deformed beam, coloured by u_z (jet) ---
        % colour per segment = average nodal dz
        c = 0.5*(dz(Tn(:,1)) + dz(Tn(:,2)));
        cmin = min(c); cmax = max(c);
        if abs(cmax-cmin) < 1e-14, cmax = cmin + 1; end

        cmap = jet(256);
        for e = 1:size(Tn,1)
            n1 = Tn(e,1); n2 = Tn(e,2);

            % map c(e) to colormap index
            t = (c(e)-cmin)/(cmax-cmin);
            idx = 1 + round(t*(size(cmap,1)-1));
            idx = max(1, min(size(cmap,1), idx));

            plot3([x(n1) x(n2)], [y(n1) y(n2)], [z(n1) z(n2)], ...
                  '-', 'Color', cmap(idx,:), 'LineWidth', 2.0);
        end

        % cosmetics
        axis equal; axis vis3d; axis off;
        view(40,20);
        %title(sprintf('Mode %d: %.2f Hz', I, freq(I)));

        colormap jet;
        cb = colorbar;
        ylabel(cb,'u_z ');
    end

    sgtitle(plotName);
end

function plotDeformed_beams(xn, Tn_st, u, scale, sigVM_st)
% Plots wing geometry (undeformed+deformed) and overlays stringers coloured by sigVM_st.
% Requires: plotDeformed.m (professor) in path, and wing.mat accessible for Tc in plotDeformed.

    % 1) Plot wing geometry using professor function (no stress)
    load('wing.mat','Tn_sk');              % uses skin as outer surface
    plotDeformed('wing', xn, Tn_sk, u, scale);  % <-- no sigVM
    hold on;

    % 2) Deformed nodal coordinates
    ux = u(1:6:end); uy = u(2:6:end); uz = u(3:6:end);
    x = xn(:,1) + scale*ux;
    y = xn(:,2) + scale*uy;
    z = xn(:,3) + scale*uz;

    % 3) Colour mapping (element-wise)
    c = sigVM_st(:);
    cmin = min(c); cmax = max(c);
    if abs(cmax-cmin) < 1e-14, cmax = cmin + 1; end
    cmap = jet(256);

    for e = 1:size(Tn_st,1)
        n1 = Tn_st(e,1); n2 = Tn_st(e,2);

        t = (c(e)-cmin)/(cmax-cmin);
        idx = 1 + round(t*(size(cmap,1)-1));
        idx = max(1, min(size(cmap,1), idx));

        plot3([x(n1) x(n2)], [y(n1) y(n2)], [z(n1) z(n2)], ...
              '-', 'Color', cmap(idx,:), 'LineWidth', 2.0);
    end

    colormap jet
    caxis([cmin cmax])  %Pa colorbar
    cb = colorbar;
    cb.Label.String = '\sigma_{VM} [Pa]';

    view(40,20);
    axis equal; axis vis3d;
end

function plotDeformed_beams(xn, Tn_st, u, scale, sigVM_st)
% plotDeformed_beams: deformed beam mesh coloured by element stress
% xn: [Nnodes x 3]
% Tn_st: [Nelem x 2]
% u: [Ndof x 1]
% scale: deformation scale
% sigVM_st: [Nelem x 1] (elementwise) or [] to colour by uz

    x0 = xn(:,1); y0 = xn(:,2); z0 = xn(:,3);

    ux = u(1:6:end); uy = u(2:6:end); uz = u(3:6:end);
    x = x0 + scale*ux;
    y = y0 + scale*uy;
    z = z0 + scale*uz;

    figure('Color','w'); hold on;

    % Undeformed discretised mesh (grey dashed)
    for e = 1:size(Tn_st,1)
        n1 = Tn_st(e,1); n2 = Tn_st(e,2);
        plot3([x0(n1) x0(n2)], [y0(n1) y0(n2)], [z0(n1) z0(n2)], ...
              ':', 'Color', 0.7*[1 1 1], 'LineWidth', 1.0);
    end

    % If stress not provided, colour by uz
    if nargin < 5 || isempty(sigVM_st)
        c = 0.5*(uz(Tn_st(:,1)) + uz(Tn_st(:,2)));   % element colour
        cb_label = 'u_z (arb.)';
    else
        c = sigVM_st(:);
        cb_label = '\sigma_{VM} [Pa]';
    end

    % Colormap mapping
    cmap = jet(256);
    cmin = min(c); 
    cmax = max(c);

    % Deformed mesh coloured
    for e = 1:size(Tn_st,1)
        n1 = Tn_st(e,1); n2 = Tn_st(e,2);
        t = (c(e)-cmin)/(cmax-cmin);
        idx = 1 + round(t*(size(cmap,1)-1));
        idx = max(1, min(size(cmap,1), idx));
        plot3([x(n1) x(n2)], [y(n1) y(n2)], [z(n1) z(n2)], ...
              '-', 'Color', cmap(idx,:), 'LineWidth', 2.0);
    end

    axis equal; axis vis3d; grid on;
    view(40,20);
    colormap jet
    caxis([cmin cmax]) %Pa
    cb = colorbar;
    cb.Label.String = '\sigma_{VM} [Pa]';
end
